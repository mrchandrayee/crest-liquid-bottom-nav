   scale, so the liquid level
   is one edge, never two */

  transform: scale(var(--k));
  transform-origin: 12px 24px;
}

/* every length is a fraction
   of ONE token: the pitch */

  --bar-h: calc(var(--pitch)
                * 1.19);
